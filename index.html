// --- NOUVELLE FONCTION : CATALOGUE HUHU.JSON --- async function loadHuhuCatalog() { document.querySelector('.source-grid').style.display = 'none'; const huhuCat = document.getElementById('huhu-catalog'); const grid = document.getElementById('huhu-grid'); huhuCat.classList.remove('hidden'); grid.innerHTML = '

Chargement du catalogue HUHU...

'; try { const res = await fetch('/api/proxy-bundle?action=catalog&type=movie&id=tmdb.movie'); const data = await res.json(); grid.innerHTML = ''; const items = data.items || []; if (items.length === 0) { grid.innerHTML = '

Aucun élément trouvé dans le catalogue.

'; return; } items.forEach(item => { const card = document.createElement('button'); card.className = 'huhu-card'; card.setAttribute('tabindex', '0'); const name = item.name || 'Sans titre'; const url = item.url || ''; card.innerHTML = `${name}`; card.onclick = async () => { if (url) { try { // On demande le vrai lien vidéo au proxy const srcRes = await fetch(`/api/proxy-bundle?action=source&type=iptv&url=${encodeURIComponent(url)}`); const srcData = await srcRes.json(); // Le vrai lien .m3u8 est souvent dans srcData.sources[0].url if (srcData.sources && srcData.sources.length > 0) { playChannel(srcData.sources[0].url, name); } else if (srcData.url) { playChannel(srcData.url, name); } else { alert("Lien vidéo introuvable pour cet élément."); } } catch (e) { alert("Erreur lors de la récupération du lien vidéo."); } } else { alert("Aucun lien vidéo disponible pour cet élément."); } }; grid.appendChild(card); }); if (grid.children.length > 0) { setTimeout(() => grid.children[0].focus(), 100); } } catch (e) { grid.innerHTML = '

Erreur de chargement du catalogue.

'; } }
